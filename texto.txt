function buildTabs() {
  const tabs = document.getElementById("categoryTabs");
  tabs.innerHTML = "";
  const all = document.createElement("button");
  all.className = "tab active";
  all.textContent = "Todos";
  all.onclick = () => filterCat(null, all);
  tabs.appendChild(all);
  cats.forEach(c => {
    const t = document.createElement("button");
    t.className = "tab";
    t.textContent = c;
    t.onclick = () => filterCat(c, t);
    tabs.appendChild(t);
  });
}

function buildMenu(items) {
  const container = document.getElementById("menuContainer");
  container.innerHTML = "";
  cats.forEach(cat => {
    const catItems = items.filter(i => i.cat === cat);
    if (!catItems.length) return;
    const title = document.createElement("div");
    title.className = "section-title";
    title.textContent = cat;
    container.appendChild(title);
    const grid = document.createElement("div");
    grid.className = "menu-grid";
    catItems.forEach(item => {
      grid.innerHTML += `
        <div class="menu-item ${cart[item.id] ? 'selected' : ''}" id="card-${item.id}" onclick="toggleItem(${item.id})">
          <div class="item-emoji">${item.emoji}</div>
          <div class="item-name">${item.name}</div>
          <div class="item-desc">${item.desc}</div>
          <div class="item-bottom">
            <div class="item-price">${formatPrice(item.price)}</div>
            <div class="item-qty" onclick="event.stopPropagation()">
              <button class="qty-btn" onclick="changeQty(${item.id},-1)">−</button>
              <span class="qty-num" id="qty-${item.id}">${cart[item.id] || 0}</span>
              <button class="qty-btn" onclick="changeQty(${item.id},1)">+</button>
            </div>
          </div>
        </div>`;
    });
    container.appendChild(grid);
  });
}

function filterCat(cat, el) {
  document.querySelectorAll(".tab").forEach(t => t.classList.remove("active"));
  el.classList.add("active");
  buildMenu(cat ? menu.filter(i => i.cat === cat) : menu);
}

function formatPrice(p) {
  return "$" + p.toLocaleString("es-CO");
}

// ══════════════════════════════════════════════════
//  CART LOGIC
// ══════════════════════════════════════════════════
function toggleItem(id) {
  if (!cart[id]) {
    cart[id] = 1;
  }
  refreshCard(id);
  updateCart();
}

function changeQty(id, delta) {
  cart[id] = Math.max(0, (cart[id] || 0) + delta);
  if (cart[id] === 0) delete cart[id];
  refreshCard(id);
  updateCart();
}

function refreshCard(id) {
  const card = document.getElementById("card-" + id);
  const qtyEl = document.getElementById("qty-" + id);
  if (!card) return;
  if (cart[id]) {
    card.classList.add("selected");
    qtyEl.textContent = cart[id];
  } else {
    card.classList.remove("selected");
    qtyEl.textContent = 0;
  }
}

function updateCart() {
  const ids = Object.keys(cart);
  const bar = document.getElementById("cartBar");
  if (!ids.length) { bar.style.display = "none"; return; }
  bar.style.display = "flex";

  let total = 0;
  let preview = "";
  ids.forEach(id => {
    const item = menu.find(m => m.id == id);
    total += item.price * cart[id];
    preview += `${item.name} × ${cart[id]}  —  ${formatPrice(item.price * cart[id])}\n`;
  });
  document.getElementById("cartPreview").textContent = preview;
  document.getElementById("cartTotal").textContent = formatPrice(total);
}

// ══════════════════════════════════════════════════
//  SUBMIT ORDER → WHATSAPP
// ══════════════════════════════════════════════════
function submitOrder() {
  const table = document.getElementById("tableNum").value.trim();
  const name  = document.getElementById("clientName").value.trim();
  if (!table) { alert("Por favor ingresa el número de mesa 🪑"); return; }
  if (!Object.keys(cart).length) { alert("Agrega algo al carrito primero 😊"); return; }

  let msg = `🍽️ *NUEVA ORDEN — Spice & Pearl*\n`;
  msg += `🪑 *Mesa:* ${table}\n`;
  msg += `👤 *Nombre:* ${name || "Anónimo"}\n\n`;
  msg += `*Pedido:*\n`;

  let total = 0;
  Object.keys(cart).forEach(id => {
    const item = menu.find(m => m.id == id);
    const subtotal = item.price * cart[id];
    total += subtotal;
    msg += `  • ${item.emoji} ${item.name} × ${cart[id]} = ${formatPrice(subtotal)}\n`;
  });
  msg += `\n💰 *TOTAL: ${formatPrice(total)}*`;

  const url = `https://wa.me/${WHATSAPP_NUMBER}?text=${encodeURIComponent(msg)}`;
  window.open(url, "_blank");
  document.getElementById("successModal").classList.add("open");
}

function resetOrder() {
  cart = {};
  document.getElementById("tableNum").value = "";
  document.getElementById("clientName").value = "";
  document.getElementById("cartBar").style.display = "none";
  document.getElementById("successModal").classList.remove("open");
  buildMenu(menu);
  window.scrollTo(0, 0);
}

function buildInventory() {
  const grid = document.getElementById("inventoryGrid");
  grid.innerHTML = "";
  inventory.forEach(item => {
    const state = invState[item.id];
    grid.innerHTML += `
      <div class="inv-item">
        <span class="inv-emoji">${item.emoji}</span>
        <span class="inv-name">${item.name}</span>
        <div class="inv-status">
          <button class="inv-btn ok ${state==='ok'?'active':''}" onclick="setInv('${item.id}','ok',this)">✓ Hay</button>
          <button class="inv-btn low ${state==='low'?'active':''}" onclick="setInv('${item.id}','low',this)">⚠ Falta</button>
        </div>
      </div>`;
  });
}

function setInv(id, state, btn) {
  invState[id] = state;
  const parent = btn.closest(".inv-status");
  parent.querySelectorAll(".inv-btn").forEach(b => b.classList.remove("active"));
  btn.classList.add("active");
}

function sendInventoryReport() {
  const falta = inventory.filter(i => invState[i.id] === "low");
  const hay   = inventory.filter(i => invState[i.id] === "ok");

  let msg = `📦 *REPORTE DE INVENTARIO — Spice & Pearl*\n`;
  msg += `📅 ${new Date().toLocaleDateString("es-CO", {weekday:"long", day:"numeric", month:"long"})}\n\n`;

  if (falta.length) {
    msg += `❌ *FALTA / Necesitas comprar:*\n`;
    falta.forEach(i => msg += `  • ${i.emoji} ${i.name}\n`);
  } else {
    msg += `✅ ¡Todo el inventario está completo!\n`;
  }

  if (hay.length && falta.length) {
    msg += `\n✅ *Disponible:*\n`;
    hay.forEach(i => msg += `  • ${i.emoji} ${i.name}\n`);
  }

  const url = `https://wa.me/${WHATSAPP_NUMBER}?text=${encodeURIComponent(msg)}`;
  window.open(url, "_blank");
}
